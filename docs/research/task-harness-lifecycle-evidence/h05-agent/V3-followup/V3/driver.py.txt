def is_interrupt_window(message, thread_id, turn_id):
 params=message.get('params',{});item=params.get('item',{})
 if message.get('method')!='item/started' or params.get('threadId')!=thread_id or params.get('turnId')!=turn_id:
  return False
 if item.get('type')!='commandExecution' or item.get('status')!='inProgress':return False
 try:
  outer=shlex.split(item.get('command',''))
  if len(outer)!=3 or outer[:2]!=['/bin/zsh','-lc']:return False
  return shlex.split(outer[2])==['python3','-B','-c',"import time; print('H05_V3_INTERRUPT_WINDOW', flush=True); time.sleep(30)"]
 except ValueError:return False
