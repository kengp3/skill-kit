import json,socket,sys,time
from pathlib import Path
base=Path(sys.argv[1]);stop=base/'listener.stop';log=base/'listener-connections.jsonl'
with socket.socket() as server:
 server.bind(('127.0.0.1',0));server.listen();server.settimeout(.2)
 port=server.getsockname()[1]
 (base/'listener-ready.json').write_text(json.dumps({'address':'127.0.0.1','port':port})+'\n')
 print(json.dumps({'listening':True,'port':port}),flush=True)
 count=0;deadline=time.monotonic()+900
 while not stop.exists() and time.monotonic()<deadline:
  try:connection,peer=server.accept()
  except socket.timeout:continue
  with connection:
   connection.settimeout(1)
   try:data=connection.recv(256).decode(errors='replace')
   except socket.timeout:data=''
  count+=1
  with log.open('a') as stream:stream.write(json.dumps({'peer':peer,'data':data})+'\n')
 (base/'listener-final.json').write_text(json.dumps({'connections':count,'stopped_by_owner':stop.exists()})+'\n')
 print(json.dumps({'connections':count,'stopped_by_owner':stop.exists()}),flush=True)
