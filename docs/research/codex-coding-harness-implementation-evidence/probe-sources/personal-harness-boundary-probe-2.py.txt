import hashlib,json,selectors,subprocess,time
from pathlib import Path
root=Path('/Users/kengp3/Workspaces/mine/simple-skills')
evidence=root/'docs/research/codex-coding-harness-implementation-evidence'
config=Path('/Users/kengp3/.codex/config.toml')
before=hashlib.sha256(config.read_bytes()).hexdigest()
command=['/Applications/ChatGPT.app/Contents/Resources/codex-cli/bin/codex',
 '-c','marketplaces.personal-harness-local.source_type="local"',
 '-c','marketplaces.personal-harness-local.source="'+str(root)+'"',
 '-c','plugins."personal-harness@personal-harness-local".enabled=false',
 'app-server','--stdio']
command=command[:1]+["app-server","--stdio"]
import tempfile,socket
workspace=Path(tempfile.mkdtemp(prefix="harness-boundary-"))
listener=socket.socket();listener.bind(("127.0.0.1",0));listener.listen(1)
port=listener.getsockname()[1]
log=[]
with (evidence/'sandbox-boundary-probe-2.stderr').open('w') as err:
 process=subprocess.Popen(command,cwd=workspace,stdin=subprocess.PIPE,stdout=subprocess.PIPE,stderr=err,text=True,bufsize=1)
 selector=selectors.DefaultSelector();selector.register(process.stdout,selectors.EVENT_READ)
 def call(id,method,params):
  request={'id':id,'method':method,'params':params};log.append({'request':request})
  process.stdin.write(json.dumps(request)+'\n');process.stdin.flush()
  deadline=time.monotonic()+30
  while time.monotonic()<deadline:
   if not selector.select(timeout=1):
    if process.poll() is not None: raise RuntimeError('app-server stopped')
    continue
   line=process.stdout.readline()
   if not line: raise RuntimeError('app-server closed stdout')
   response=json.loads(line);log.append({'response':response})
   if response.get('id')==id:
    if 'error' in response: raise RuntimeError(str(response['error']))
    return response.get('result')
  raise TimeoutError(method)
 try:
  call(1,'initialize',{'clientInfo':{'name':'personal_harness_probe','version':'0.1.0'},'capabilities':{'experimentalApi':True}})
  process.stdin.write(json.dumps({'method':'initialized'})+'\n');process.stdin.flush()
  import tempfile,sys
  outside=root/'__harness_boundary_canary__'
  assert not outside.exists()
  policy={'type':'workspaceWrite','writableRoots':[],'excludeSlashTmp':True,'excludeTmpdirEnvVar':True,'networkAccess':False}
  def execute(id,code,extra=[]):
   return call(id,'command/exec',{'cwd':str(workspace),'command':[sys.executable,'-c',code,*extra],'sandboxPolicy':policy,'timeoutMs':5000})
  allowed=execute(2,'from pathlib import Path; Path("allowed").write_text("ok"); print("allowed")')
  denied=execute(3,'from pathlib import Path; import sys; Path(sys.argv[1]).write_text("must be denied")',[str(outside)])
  network=execute(4,'import socket,sys; socket.create_connection(("127.0.0.1",int(sys.argv[1])),timeout=1).send(b"probe")',[str(port)])
  (evidence/'sandbox-boundary-observation-2.json').write_text(json.dumps({'workspace':str(workspace),'allowed_marker':(workspace/'allowed').exists(),'original_repo_marker_exists':outside.exists(),'allowed_exit':allowed['exitCode'],'denied_exit':denied['exitCode'],'network_exit':network['exitCode'],'server_cwd':str(workspace)},indent=2)+'\n')
  safe=(workspace/'allowed').exists() and not outside.exists() and denied['exitCode']!=0 and network['exitCode']!=0
  if outside.exists(): outside.unlink()
  assert safe, 'effective sandbox boundary did not match trial requirements'

 finally:
  process.terminate()
  try: process.wait(timeout=5)
  except subprocess.TimeoutExpired: process.kill();process.wait()
  after=hashlib.sha256(config.read_bytes()).hexdigest()
  (evidence/'sandbox-boundary-probe-2.json').write_text(json.dumps({'argv':command,'user_config_before':before,'user_config_after':after,'events':log},indent=2)+'\n')
print('events',len(log),'user config unchanged',before==after)
